        /* =========================================================
           설정
        ========================================================= */

        const TOTAL_STUDENTS = 30;
        const SELECT_COUNT = 5;


        /* =========================================================
           DOM
        ========================================================= */

        const drawButton =
            document.getElementById("drawButton");

        const resultContainer =
            document.getElementById("resultContainer");

        const drawCountElement =
            document.getElementById("drawCount");


        /* =========================================================
           상태
        ========================================================= */

        let drawCount = 0;


        /* =========================================================
           랜덤 추첨 함수
           
           Fisher-Yates Shuffle을 사용하여
           1~30번 중 중복 없이 5명을 선정
        ========================================================= */

        function drawStudents() {

            // 1 ~ 30 생성
            const students =
                Array.from(
                    { length: TOTAL_STUDENTS },
                    (_, index) => index + 1
                );

            // 배열 섞기
            for (
                let i = students.length - 1;
                i > 0;
                i--
            ) {

                const randomIndex =
                    Math.floor(
                        Math.random() * (i + 1)
                    );

                [
                    students[i],
                    students[randomIndex]
                ] = [
                    students[randomIndex],
                    students[i]
                ];
            }

            // 5명 선택
            return students.slice(
                0,
                SELECT_COUNT
            );
        }


        /* =========================================================
           결과 화면 출력
        ========================================================= */

        function renderResults(selectedStudents) {

            resultContainer.innerHTML = "";

            const grid =
                document.createElement("div");

            grid.className =
                "student-grid";


            selectedStudents.forEach(
                (studentNumber, index) => {

                    const card =
                        document.createElement("div");

                    card.className =
                        "student-card";

                    card.style.animationDelay =
                        `${index * 0.12}s`;


                    card.innerHTML = `

                        <div class="student-rank">
                            ${index + 1}
                        </div>

                        <div class="student-number">
                            ${studentNumber}
                        </div>

                        <div class="student-label">
                            번 학생
                        </div>

                    `;

                    grid.appendChild(card);
                }
            );


            resultContainer.appendChild(grid);
        }

        async function startDrawAnimation() {

            drawButton.disabled = true;

            drawButton.innerHTML = `
                <span
                    class="spinner-border spinner-border-sm me-2"
                    role="status"
                ></span>

                추첨 중...
            `;


            // 추첨 준비 SweetAlert
            await Swal.fire({

                title: "🎲 추첨을 시작합니다!",
                html: `
                    <div class="text-muted">
                        공정한 랜덤 시스템으로<br>
                        5명을 선정하고 있습니다.
                    </div>
                `,

                timer: 1200,

                showConfirmButton: false,

                allowOutsideClick: false,

                background: "#ffffff",

                icon: "info"
            });


            // 실제 추첨
            const selectedStudents =
                drawStudents();


            // 결과 표시
            renderResults(
                selectedStudents
            );


            drawCount++;

            drawCountElement.textContent =
                drawCount;


            // 버튼 복구
            drawButton.disabled = false;

            drawButton.innerHTML = `
                <i class="bi bi-shuffle me-2"></i>
                다시 뽑기
            `;


            // 결과 알림
            await Swal.fire({

                title: "🎉 청소당번 선정 완료!",

                html: `
                    <div
                        style="
                            font-size: 1.1rem;
                            line-height: 2;

                    >
                        오늘의 청소당번은
                        <strong

function tone(freq,dur=.08,type="sine",gain=.02,delay=0){
  if(!audioCtx)return;

  try{
    const o=audioCtx.createOscillator();
    const g=audioCtx.createGain();

    o.type=type;
    o.frequency.value=freq;

    const t=audioCtx.currentTime+delay;

    g.gain.setValueAtTime(.0001,t);
    g.gain.exponentialRampToValueAtTime(gain,t+.01);
    g.gain.exponentialRampToValueAtTime(.0001,t+dur);

    o.connect(g);
    g.connect(audioCtx.destination);

    o.start(t);
    o.stop(t+dur+.02);
  }catch(e){}
}

function parseNames(){

    .split(/\r?\n/)
    .map(x=>x.trim())
    .filter(Boolean);

  const result=[];

  for(const line of lines){
    const m=line.match(/^(.*?)[*xX×]\s*(\d+)$/);

    if(m){
      const name=m[1].trim();
      const count=Number(m[2]);

      if(!name||!Number.isFinite(count)||count<1||count>100){
        return null;
      }

      for(let i=0;i<count;i++){
        result.push(name);
      }
    }else{
      result.push(line);
    }
  }

  if(result.length<2||result.length>100)return null;
  return result;
}

function resolveSegmentBall(b,o){
  const x1=o.x;
  const y1=o.y;

  const x2=o.x+o.w;
  const y2=o.y+
    (o.dir>0?o.w*.27:-o.w*.27);

  const dx=x2-x1;
  const dy=y2-y1;
  const len2=dx*dx+dy*dy;

  let t=
    ((b.x-x1)*dx+(b.y-y1)*dy)/
    (len2||1);

  t=clamp(t,0,1);

  const px=x1+dx*t;
  const py=y1+dy*t;

  let nx=b.x-px;
  let ny=b.y-py;

  let dist=Math.hypot(nx,ny);

  if(dist<.0001){
    nx=-dy;
    ny=dx;
    dist=Math.hypot(nx,ny)||1;
  }

  nx/=dist;
  ny/=dist;

  const radius=b.r+o.h*.5;

  if(dist<radius){
    const push=radius-dist+.9;

    b.x+=nx*push;
    b.y+=ny*push;

    const vn=b.vx*nx+b.vy*ny;

    if(vn<0){
      b.vx-=vn*1.65*nx;
      b.vy-=vn*1.65*ny;

      b.vx*=.94;
      b.vy*=.94;
    }

    const len=Math.hypot(dx,dy)||1;
    const tx=dx/len;
    const ty=dy/len;

    b.vx+=tx*16;
    b.vy+=Math.abs(ty)*10;

    b.glow=.8;

    return true;
  }

  return false;
}

function resolveFinishSlope(b,o){
  const dx=o.x2-o.x1;
  const dy=o.y2-o.y1;

  const len2=dx*dx+dy*dy;

  let t=
    ((b.x-o.x1)*dx+(b.y-o.y1)*dy)/
    (len2||1);

  t=clamp(t,0,1);

  const px=o.x1+dx*t;
  const py=o.y1+dy*t;

  let nx=b.x-px;
  let ny=b.y-py;

  const d=Math.hypot(nx,ny)||.001;

  nx/=d;
  ny/=d;

  const minDist=b.r+o.h/2;

  if(d<minDist){
    const penetration=minDist-d+1;

    b.x+=nx*penetration;
    b.y+=ny*penetration;

    const vn=b.vx*nx+b.vy*ny;

    if(vn<0){
      b.vx-=vn*1.75*nx;
      b.vy-=vn*1.75*ny;
    }

    b.glow=.8;
    return true;
  }

  return false;
}

function resolveCircle(b,o){
  // 이미 다른 구슬이 먹은 아이템은 충돌 처리하지 않는다.
  if(
    (o.type==="push"||o.type==="jump") &&
    o.active===false
  ){
    return false;
  }

  const dx=b.x-o.x;
  const dy=b.y-o.y;
  const d=Math.hypot(dx,dy)||.001;

  const min=b.r+o.r;

  if(d>=min)return false;

  const nx=dx/d;
  const ny=dy/d;
  const penetration=min-d+1;

  b.x+=nx*penetration;
  b.y+=ny*penetration;

  const vn=b.vx*nx+b.vy*ny;

  if(vn<0){
    b.vx-=vn*1.8*nx;
    b.vy-=vn*1.8*ny;
  }

  if(o.type==="push"){
    if(!b.itemFlags.push){

      // ★ 핵심 수정:
      // 첫 번째 구슬이 먹는 순간 아이템 자체를 제거
      o.active=false;
      b.itemFlags.push=true;

      const force=300;

      b.vx+=nx*force;
      b.vy+=ny*force;

      );
    }
  }else{
    b.glow=1;
    o.pulse=1;

    if(Math.random()<.18){
      hitSound();
    }
  }

  return true;
}

/* =========================
   ROTATING FINAL BAR
========================= */

function resolveRotatingBar(b){
  const cx=rotatingBar.x;
  const cy=rotatingBar.y;

  const cos=Math.cos(rotatingBar.angle);
  const sin=Math.sin(rotatingBar.angle);

  const half=rotatingBar.length/2;

  const ax=cx-cos*half;
  const ay=cy-sin*half;

  const bx=cx+cos*half;
  const by=cy+sin*half;

  const dx=bx-ax;
  const dy=by-ay;
  const len2=dx*dx+dy*dy;

  let t=
    ((b.x-ax)*dx+(b.y-ay)*dy)/
    (len2||1);

  t=clamp(t,0,1);

  const px=ax+dx*t;
  const py=ay+dy*t;

  let nx=b.x-px;
  let ny=b.y-py;

  const d=Math.hypot(nx,ny)||.001;

  nx/=d;
  ny/=d;

  const minDist=b.r+rotatingBar.thickness/2;

  if(d<minDist){
    const penetration=minDist-d+1;

    b.x+=nx*penetration;
    b.y+=ny*penetration;

    const rx=px-cx;
    const ry=py-cy;

    const barVX=-rotatingBar.speed*ry;
    const barVY= rotatingBar.speed*rx;

    const relVX=b.vx-barVX;
    const relVY=b.vy-barVY;

    const vn=relVX*nx+relVY*ny;

    if(vn<0){
      b.vx-=vn*1.9*nx;
      b.vy-=vn*1.9*ny;
    }

    b.vx+=barVX*.22;
    b.vy+=barVY*.22;

    b.glow=1;

    if(Math.random()<.1)hitSound();
  }
}

function applyWalls(b){
  const wall=19;

  if(b.x-b.r<wall){
    b.x=wall+b.r+.5;

    if(b.vx<0){
      b.vx=-b.vx*.86;
    }

    b.glow=.5;
  }

  if(b.x+b.r>W-wall){
    b.x=W-wall-b.r-.5;

    if(b.vx>0){
      b.vx=-b.vx*.86;
    }

    b.glow=.5;
  }
}

function applyJumpItem(b,o){
  // 이미 사용된 점프패드는 완전히 비활성화
  if(o.active===false)return;

  if(
    b.x>o.x-o.w/2-b.r &&
    b.x<o.x+o.w/2+b.r &&
    b.y>o.y-o.h-b.r &&
    b.y<o.y+o.h+b.r
  ){
    if(!b.itemFlags.jump){

      // ★ 핵심 수정:
      // 첫 구슬이 밟는 즉시 점프패드 제거
      o.active=false;
      b.itemFlags.jump=true;

      b.vy=-Math.max(
        420,
        Math.abs(b.vy)*1.55
      );

      b.vx+=rand(-80,80);
      b.jump=.35;

      burst(
        b.x,b.y,
        "#ffd95a",
        12
      );

      showEvent(
        "JUMP!",
        b.name+" 점프패드 사용",
        "#ffd95a"
      );

      hitSound();
    }
  }
}

function applyObstacles(b){
  applyWalls(b);

  for(const o of obstacles){

    if(
      o.type==="circle"||
      o.type==="push"
    ){
      resolveCircle(b,o);
    }

    else if(o.type==="slope"){
      resolveSegmentBall(b,o);
    }

    else if(o.type==="finishSlope"){
      resolveFinishSlope(b,o);
    }

    else if(o.type==="jump"){
      applyJumpItem(b,o);
    }
  }

  if(
    b.y>
    rotatingBar.y-rotatingBar.length/2-35 &&
    b.y<
    rotatingBar.y+rotatingBar.length/2+35
  ){
    resolveRotatingBar(b);
  }
}

/* =========================
   RACER COLLISION

function collideRacers(){
  for(let i=0;i<racers.length;i++){
    const a=racers[i];

    if(!a.alive||a.finished)continue;

    for(let j=i+1;j<racers.length;j++){
      const b=racers[j];

      if(!b.alive||b.finished)continue;

      const dx=b.x-a.x;
      const dy=b.y-a.y;

      const d=Math.hypot(dx,dy)||.001;
      const min=a.r+b.r;

      if(d<min){
        const nx=dx/d;
        const ny=dy/d;

        const overlap=min-d+.7;

        a.x-=nx*overlap*.5;
        a.y-=ny*overlap*.5;

        b.x+=nx*overlap*.5;
        b.y+=ny*overlap*.5;

        const rv=
          (b.vx-a.vx)*nx+
          (b.vy-a.vy)*ny;

        if(rv<0){
          const impulse=-rv*.68;

          a.vx-=nx*impulse;
          a.vy-=ny*impulse;

          b.vx+=nx*impulse;
          b.vy+=ny*impulse;
        }

        a.vx-=nx*3;
        a.vy-=ny*2;

        b.vx+=nx*3;
        b.vy+=ny*2;
      }
    }
  }
}

/* =========================
   PHYSICS
========================= */

function updateRacer(b,dt){
  const oldY=b.y;

  b.vy+=310*dt;

  b.vx+=
    Math.sin(b.y*.006+b.seed)*
    18*dt;

  b.vx+=
    Math.sin(b.y*.019+b.seed*.73)*
    9*dt;

  b.vx*=Math.pow(.994,dt*60);
  b.vy*=Math.pow(.998,dt*60);

  if(b.jump>0){
    b.jump-=dt;
    b.vy-=80*dt;
  }

  b.x+=b.vx*dt;
  b.y+=b.vy*dt;

  for(let i=0;i<3;i++){
    applyObstacles(b);
  }

  const moved=b.y-oldY;

  if(moved<.25){
    b.stuck+=dt;
  }else{
    b.stuck=Math.max(
      0,
      b.stuck-dt*.7
    );
  }

  if(b.stuck>.55){
    b.vy=Math.max(
      b.vy,
      rand(130,220)
    );

    b.vx+=rand(-130,130);
    b.y+=4;

    b.stuck=0;
  }

  if(Math.hypot(b.vx,b.vy)<55){
    b.vy+=45;
    b.vx+=rand(-10,10);
  }

  b.vx=clamp(
    b.vx,
    -500,
    500
  );

  b.vy=clamp(
    b.vy,
    -650,
    650
  );

  b.trail.push({
    x:b.x,
    y:b.y
  });

  if(b.trail.length>7){
    b.trail.shift();
  }

  b.glow=Math.max(
    0,
    b.glow-dt*2.8
  );
}

/* =========================
   FINISH
========================= */

function processFinish(){
  if(finished)return;

  const finalY=WORLD_H-105;

  for(const b of racers){
    if(
      b.alive &&
      !b.finished &&
      b.y>finalY &&
      Math.abs(b.x-W/2)<32
    ){
      b.finished=true;

      finishers.push(b);

      b.vx=0;
      b.vy=0;

      b.x=W/2;
      b.y=WORLD_H-70;
    }
  }

  if(finishers.length){
    finishTimer-=1/60;

    if(finishTimer<=0){
      const b=finishers.shift();

      if(b){
        b.alive=false;

        b.rank=
          racers.filter(
            x=>x.rank>0
          ).length+1;

        burst(
          W/2,
          WORLD_H-70,
          "#ffd95a",
          18
        );

        showEvent(
          b.rank+"위",
          b.name+" 결승 통과!",
          "#ffd95a"
        );

        finishTimer=.38;
      }
    }
  }

  const alive=
    racers.filter(r=>r.alive);

  if(alive.length===1){
    const last=alive[0];

    if(last.y>WORLD_H-350){
      last.vx+=
        (W/2-last.x)*.5;

      last.vy=
        Math.max(
          last.vy,
          130
        );
    }
  }

  if(
    racers.every(
      r=>!r.alive
    )
  ){
    const winner=
      racers
        .filter(r=>r.rank>0)
        .sort(
          (a,b)=>a.rank-b.rank
        )[0];

    if(winner){
      finishGame(winner);
    }
  }
}

}

/* =========================
   PARTICLES
========================= */

function burst(x,y,color,n){
  for(let i=0;i<n;i++){
    if(particles.length>500)break;

    const a=Math.random()*Math.PI*2;
    const s=rand(50,220);

    particles.push({
      x,y,
      vx:Math.cos(a)*s,
      vy:Math.sin(a)*s,
      life:rand(.3,.8),
      max:.8,
      color,
      size:rand(1,3.5)
    });
  }
}

function updateParticles(dt){
  for(
    let i=particles.length-1;
    i>=0;
    i--
  ){
    const p=particles[i];

    p.life-=dt;
    p.vy+=220*dt;

    p.x+=p.vx*dt;
    p.y+=p.vy*dt;

    if(p.life<=0){
      particles.splice(i,1);
    }
  }
}

/* =========================
   CAMERA
========================= */

function updateCamera(){
  const alive=
    racers.filter(
      r=>r.alive&&!r.finished
    );

  if(!alive.length)return;

  const finalY=WORLD_H-550;

  const nearFinish=
    alive
      .filter(r=>r.y>finalY)
      .sort((a,b)=>b.y-a.y);

  let target=null;

  if(nearFinish.length){
    target=nearFinish[0];
    cameraInfo.textContent="결승 추적";
  }

  if(!target){
    const sorted=
      [...alive]
        .sort((a,b)=>b.y-a.y);

    target=
      sorted[
        Math.floor(
          sorted.length*.48
        )
      ]||sorted[0];

    cameraInfo.textContent="중간 추적";
  }

  let desired=
    target.y-H*.52;

  if(target.y>finalY){
    desired=
      target.y-H*.60;
  }

  cameraY=lerp(
    cameraY,
    clamp(
      desired,
      0,
      WORLD_H-H
    ),
    .09
  );
}

/* =========================
   MAIN LOOP
========================= */

function loop(t){
  const dt=
    Math.min(
      .032,
      (t-lastTime)/1000||.016
    );

  lastTime=t;

  if(active){

    const substeps=
      racers.length>70
      ?4
      :3;

    const sdt=
      dt/substeps;

    for(
      let s=0;
      s<substeps;
      s++
    ){

      rotatingBar.angle+=
        rotatingBar.speed*sdt;

      for(const b of racers){
        if(
          b.alive&&
          !b.finished
        ){
          updateRacer(
            b,
            sdt
          );
        }
      }

      collideRacers();

      for(const b of racers){
        if(
          b.alive&&
          !b.finished&&
          b.y>WORLD_H+80
        ){
          eliminate(b);
        }
      }

      processFinish();
    }

    updateCamera();
    updateHUD();
  }

  updateParticles(dt);
